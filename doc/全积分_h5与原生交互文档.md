## 全积分 h5与原生交互文档
``` js
        // 检查原生接口是否可用
        function checkNativeInterface() {
            if (window.JSNative) {
                //原生接口已准备好
                return true;
            } else {
                //原生接口不可用
                return false;
            }
        }
        
        //打开Chrome id任务id url链接
        function openChrome(id,url) {
            if (!checkNativeInterface()) return;
            var message = JSON.stringify({
                id:id,
                action: "open",
                url: url
            });
            window.JSNative.postMessage(message);
        }
        
        
        //原生关闭chrome回调 id任务id code为0表示执行成功其他为失败
        function onChromeClose(result) {
           //result "{id:id, code:0, msg:"success"}"
        }
```
